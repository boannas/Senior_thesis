\begin{algorithm}[H]
\caption{Agent Update Cycle at Time $t$}
\begin{algorithmic}[1]

\State \textbf{Input:} Environment state $E(t)$, agent state $s_i(t)$, genotype $G_i$
\State \textbf{Output:} Updated state $s_i(t+1)$ and environment $E(t+1)$

\State $o_i(t) \gets P_i(E(t))$ \Comment{Perception: extract cues}
\State $m_i(t) \gets M_i(G_i, s_i(t), o_i(t))$ \Comment{Motivation update}
\State $a_i(t) \gets \pi_i(G_i, m_i(t), s_i(t), o_i(t))$ \Comment{Choose action}
\State $s_i(t+1) \gets f_i(G_i, s_i(t), o_i(t), a_i(t))$ \Comment{Internal update}
\State $E(t+1) \gets g(E(t), a_1(t), \ldots, a_K(t))$ \Comment{Environment update}

\State \Return $(s_i(t+1), E(t+1))$

\end{algorithmic}
\end{algorithm}
